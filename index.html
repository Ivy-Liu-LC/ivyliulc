function normalizeCellChar(value) {
if (typeof value !== "string" || value.length === 0) {
return " ";
}

return value.charAt(0);
}

function safeTimestamp(value) {
var numberValue = Number(value);

if (!isFinite(numberValue) || numberValue < 0) {
return 0;
}

return Math.floor(numberValue);
}

// Build a stable key for a world-space character cell.
function mapKey(worldX, worldY) {
return String(worldX) + "," + String(worldY);
}

function parseWorldKey(key) {
var comma;
var worldX;
var worldY;

if (typeof key !== "string") {
return null;
}

comma = key.indexOf(",");

if (comma <= 0 || comma >= key.length - 1) {
return null;
}

worldX = parseInt(key.slice(0, comma), 10);
worldY = parseInt(key.slice(comma + 1), 10);

if (!isFinite(worldX) || !isFinite(worldY)) {
return null;
}

return {
worldX: worldX,
worldY: worldY,
key: mapKey(worldX, worldY),
};
}

if (cellCountNode) {

function buildLocalSnapshot() {
var snapshot = { cells: {}, updatedAt: 0 };

versionMap.forEach(function (timestamp, key) {
var safeTime = safeTimestamp(timestamp);
var currentValue = cellMap.has(key) ? cellMap.get(key) : " ";

snapshot.cells[key] = {
ch: normalizeCellChar(currentValue),
updatedAt: safeTime,
};

if (safeTime > snapshot.updatedAt) {
snapshot.updatedAt = safeTime;
}
});

return snapshot;
}

function normalizeRemoteDocument(raw) {
var output = { cells: {}, updatedAt: 0 };
var key;

if (!raw || typeof raw !== "object") {
return output;
}

output.updatedAt = safeTimestamp(raw.updatedAt);

if (!raw.cells || typeof raw.cells !== "object" || Array.isArray(raw.cells)) {
return output;
}

for (key in raw.cells) {
var parsedKey;
var item;
var ch;
var updatedAt;

if (!Object.prototype.hasOwnProperty.call(raw.cells, key)) {
continue;
}

parsedKey = parseWorldKey(key);

if (!parsedKey) {
continue;
}

item = raw.cells[key];

if (!item || typeof item !== "object") {
continue;
}

ch = normalizeCellChar(item.ch);
updatedAt = safeTimestamp(item.updatedAt);

output.cells[parsedKey.key] = {
ch: ch,
updatedAt: updatedAt,
};

if (updatedAt > output.updatedAt) {
output.updatedAt = updatedAt;
}
}

return output;
}

function mergeSnapshots(remoteSnapshot, localSnapshot) {
var merged = { cells: {}, updatedAt: 0 };
var allKeys = {};
var key;

for (key in remoteSnapshot.cells) {
if (Object.prototype.hasOwnProperty.call(remoteSnapshot.cells, key)) {
allKeys[key] = true;
}
}

for (key in localSnapshot.cells) {
if (Object.prototype.hasOwnProperty.call(localSnapshot.cells, key)) {
allKeys[key] = true;
}
}

for (key in allKeys) {
var remoteCell = remoteSnapshot.cells[key];
var localCell = localSnapshot.cells[key];
var chosen;
var chosenTime;
var remoteDocTime;

if (!Object.prototype.hasOwnProperty.call(allKeys, key)) {
continue;
}

if (remoteCell && localCell) {
chosen =
safeTimestamp(localCell.updatedAt) >= safeTimestamp(remoteCell.updatedAt)
? localCell
: remoteCell;
} else if (!remoteCell && localCell) {
remoteDocTime = safeTimestamp(remoteSnapshot.updatedAt);
chosen =
remoteDocTime > safeTimestamp(localCell.updatedAt)
? { ch: " ", updatedAt: remoteDocTime }
: localCell;
} else {
chosen = localCell || remoteCell;
}

if (!chosen) {
continue;
}

chosenTime = safeTimestamp(chosen.updatedAt);

merged.cells[key] = {
ch: normalizeCellChar(chosen.ch),
updatedAt: chosenTime,
};

if (chosenTime > merged.updatedAt) {
merged.updatedAt = chosenTime;
}
}

return merged;
}

function applySnapshot(snapshot) {
var key;
var remoteDocTime = safeTimestamp(snapshot.updatedAt);

for (key in snapshot.cells) {
var incoming;
var incomingTime;
var localTime;

if (!Object.prototype.hasOwnProperty.call(snapshot.cells, key)) {
continue;
}

incoming = snapshot.cells[key];
incomingTime = safeTimestamp(incoming.updatedAt);
localTime = safeTimestamp(versionMap.get(key));

if (incomingTime < localTime) {
continue;
}

versionMap.set(key, incomingTime);

if (normalizeCellChar(incoming.ch) === " ") {
cellMap.delete(key);
} else {
cellMap.set(key, normalizeCellChar(incoming.ch));
}
}

if (remoteDocTime > 0) {
versionMap.forEach(function (timestamp, localKey) {
if (
!Object.prototype.hasOwnProperty.call(snapshot.cells, localKey) &&
safeTimestamp(timestamp) < remoteDocTime
) {
versionMap.set(localKey, remoteDocTime);
cellMap.delete(localKey);
}
});
}

updateCellCount();
}

// Write one character to the world map and schedule persistence.
function setCell(worldX, worldY, value) {
var key = mapKey(worldX, worldY);
var normalized = normalizeCellChar(value);
var previous = cellMap.has(key) ? cellMap.get(key) : " ";
var now = Date.now();

if (normalized === previous) {
return;
}

if (normalized === " ") {
cellMap.delete(key);
} else {
cellMap.set(key, normalized);
}

versionMap.set(key, now);
updateCellCount();

if (localSaveTimer) {
window.clearTimeout(localSaveTimer);
}

localSaveTimer = window.setTimeout(function () {
localSaveTimer = 0;
saveToStorage();
}, 300);

scheduleRemoteSync(1800);
}

function loadLegacyArray(parsedArray) {
var idx;

if (!Array.isArray(parsedArray)) {
return;
}

for (idx = 0; idx < parsedArray.length; idx += 1) {
var item = parsedArray[idx];
var parsedKey;

if (!Array.isArray(item) || item.length < 2) {
continue;
}

parsedKey = parseWorldKey(item[0]);

if (!parsedKey || typeof item[1] !== "string") {
continue;
}

if (item[1].length === 1 && item[1] !== " ") {
cellMap.set(parsedKey.key, item[1]);
versionMap.set(parsedKey.key, safeTimestamp(item[2]));
}
}

updateCellCount();
}

// Persist all non-empty world cells to localStorage.
function saveToStorage() {
var snapshot = buildLocalSnapshot();

try {
localStorage.setItem(guestbookStorageKey, JSON.stringify(snapshot));
} catch (error) {
// Ignore quota/storage errors.
}
}

// Restore previously saved world cells from localStorage.
function loadFromStorage() {
var raw;
var parsed;
var idx;

try {
raw = localStorage.getItem(guestbookStorageKey);
} catch (error) {
return;
}

if (!raw) {
return;
}

try {
parsed = JSON.parse(raw);
} catch (error) {
return;
}

if (Array.isArray(parsed)) {
loadLegacyArray(parsed);
return;
}

if (!parsed || typeof parsed !== "object") {
return;
}

if (Array.isArray(parsed.cells)) {
loadLegacyArray(parsed.cells);
return;
}

if (!parsed.cells || typeof parsed.cells !== "object") {
return;
}

for (idx = 0; idx < Object.keys(parsed.cells).length; idx += 1) {
var key = Object.keys(parsed.cells)[idx];
var parsedKey = parseWorldKey(key);
var item = parsed.cells[key];
var ch;
var updatedAt;

if (!parsedKey || !item || typeof item !== "object") {
continue;
}

ch = normalizeCellChar(item.ch);
updatedAt = safeTimestamp(item.updatedAt);
versionMap.set(parsedKey.key, updatedAt);

if (ch === " ") {
cellMap.delete(parsedKey.key);
} else {
cellMap.set(parsedKey.key, ch);
}
}

updateCellCount();
}

async function fetchRemoteSnapshot() {
var response;
var payload;

response = await fetch(guestbookSharedEndpoint, {
method: "GET",
cache: "no-store",
});

if (!response.ok) {
throw new Error("Remote read failed: " + response.status);
}

payload = await response.json();

return {
etag: response.headers.get("etag"),
snapshot: normalizeRemoteDocument(payload),
};
}

async function writeRemoteSnapshot(snapshot, etag) {
var response;
var headers = {
"Content-Type": "application/json",
};

if (etag) {
headers["If-Match"] = etag;
}

response = await fetch(guestbookSharedEndpoint, {
method: "PUT",
headers: headers,
body: JSON.stringify(snapshot),
});

if (response.status === 412) {
var conflictError = new Error("Conflict");
conflictError.code = 412;
throw conflictError;
}

if (!response.ok) {
throw new Error("Remote write failed: " + response.status);
}
}

function scheduleRemoteSync(delay) {

return;

async function syncRemote(force) {
var now = Date.now();
var attempts = 0;
var writeReadyAt;

if (!guestbookSharedEndpoint) {
setSyncStatus("local only");
return;
}

if (remoteSyncInFlight) {
remoteSyncQueued = true;
return;
}

writeReadyAt = Math.max(0, nextAllowedRemoteWriteAt - now);

if (!force && writeReadyAt > 0) {
scheduleRemoteSync(writeReadyAt);
return;
}

remoteSyncInFlight = true;
setSyncStatus("syncing");

while (attempts < 3) {
var remoteState;
var localSnapshot;
var mergedSnapshot;

attempts += 1;

try {
remoteState = await fetchRemoteSnapshot();
localSnapshot = buildLocalSnapshot();
mergedSnapshot = mergeSnapshots(remoteState.snapshot, localSnapshot);
await writeRemoteSnapshot(mergedSnapshot, remoteState.etag);
applySnapshot(mergedSnapshot);
saveToStorage();
setSyncStatus("synced");
nextAllowedRemoteWriteAt = Date.now() + remoteMinWriteIntervalMs;
remoteSyncInFlight = false;

if (remoteSyncQueued) {
remoteSyncQueued = false;
scheduleRemoteSync(remoteMinWriteIntervalMs);
}

return;
} catch (error) {
if (!error || error.code !== 412) {
setSyncStatus("local only");
nextAllowedRemoteWriteAt = Date.now() + remoteMinWriteIntervalMs;
remoteSyncInFlight = false;

if (remoteSyncQueued) {
remoteSyncQueued = false;
scheduleRemoteSync(remoteMinWriteIntervalMs);
}

return;
}
}
}

setSyncStatus("sync conflict");
remoteSyncInFlight = false;
scheduleRemoteSync(remoteMinWriteIntervalMs);
}

async function pullRemoteSnapshot() {
if (!guestbookSharedEndpoint) {
return;
}

try {
var remoteState = await fetchRemoteSnapshot();
applySnapshot(remoteState.snapshot);
saveToStorage();

if (!remoteSyncInFlight) {
setSyncStatus("synced");
}
} catch (error) {
if (!remoteSyncInFlight) {
setSyncStatus("local only");
}
}
}

// Convert a mouse position on screen into absolute world cell coordinates.
function clientToWorld(clientX, clientY) {
var rect = canvas.getBoundingClientRect();
var screenX = clientX - rect.left;
var screenY = clientY - rect.top;

return {
worldX: Math.floor((screenX + offsetX) / cellWidth),
worldY: Math.floor((screenY + offsetY) / cellHeight),
};
}

function moveCursor(dx, dy) {
cursorX += dx;
cursorY += dy;
}

function insertTextAtCursor(text) {
var idx;

if (!cursorActive) {
return;
}

for (idx = 0; idx < text.length; idx += 1) {
var ch = text.charAt(idx);

if (ch === "\r") {
continue;
}

if (ch === "\n") {
cursorX = guestbookStartPoint.x;
cursorY += 1;
continue;
}

setCell(cursorX, cursorY, ch);
moveCursor(1, 0);
}
}

// Trackpad/mouse-wheel panning updates the camera offset.
function onWheel(event) {
event.preventDefault();
offsetX += event.deltaX;
offsetY += event.deltaY;
}

// Start drag-panning with mouse or touch.
function onPointerDown(event) {
if (event.pointerType === "mouse" && event.button !== 0) {
return;
}

dragging = true;
moved = false;
activePointerId = event.pointerId;
lastDragX = event.clientX;
lastDragY = event.clientY;
canvas.classList.add("dragging");
if (typeof canvas.setPointerCapture === "function") {
canvas.setPointerCapture(event.pointerId);
}
canvas.focus();
mobileInput.focus();
event.preventDefault();
}

// Continue drag-panning while pointer is held.
function onPointerMove(event) {
var dx;
var dy;

if (!dragging || activePointerId !== event.pointerId) {
return;
}

dx = event.clientX - lastDragX;
dy = event.clientY - lastDragY;

if (dx !== 0 || dy !== 0) {
moved = true;
}

offsetX -= dx;
offsetY -= dy;
lastDragX = event.clientX;
lastDragY = event.clientY;
event.preventDefault();
}

// End dragging; if no movement happened, treat it as click/tap to place cursor.
function onPointerUp(event) {
var hit;

if (!dragging || activePointerId !== event.pointerId) {
return;
}

dragging = false;
activePointerId = null;
canvas.classList.remove("dragging");

if (!moved) {
hit = clientToWorld(event.clientX, event.clientY);
cursorX = hit.worldX;
cursorY = hit.worldY;
cursorActive = true;
canvas.focus();
mobileInput.focus();
}

event.preventDefault();
}

function onPointerCancel(event) {
if (activePointerId !== event.pointerId) {
return;
}

dragging = false;
activePointerId = null;
canvas.classList.remove("dragging");
}

// Keyboard editing writes directly into world cells and moves cursor.
function onKeyDown(event) {
if (!cursorActive) {
return;
}

if (event.metaKey || event.ctrlKey) {
return;
}

if (event.key === "Backspace") {
event.preventDefault();
moveCursor(-1, 0);
setCell(cursorX, cursorY, "");
return;
}

if (event.key === "Delete") {
event.preventDefault();
setCell(cursorX, cursorY, "");
return;
}

if (event.key === "Enter") {
event.preventDefault();
cursorX = guestbookStartPoint.x;
cursorY += 1;
return;
}

if (event.key === "ArrowLeft") {
event.preventDefault();
moveCursor(-1, 0);
return;
}

if (event.key === "ArrowRight") {
event.preventDefault();
moveCursor(1, 0);
return;
}

if (event.key === "ArrowUp") {
event.preventDefault();
moveCursor(0, -1);
return;
}

if (event.key === "ArrowDown") {
event.preventDefault();
moveCursor(0, 1);
return;
}

if (event.key === "Tab") {
event.preventDefault();
moveCursor(2, 0);
return;
}

if (event.key.length === 1 && !event.altKey) {
event.preventDefault();
insertTextAtCursor(event.key);
}
}

// Keep the canvas resolution in sync with window size and device pixel ratio.
function resizeCanvas() {

var m;

rect = canvas.getBoundingClientRect();
canvas.width = Math.max(1, Math.floor(rect.width * dpr));
canvas.height = Math.max(1, Math.floor(rect.height * dpr));

}

// Render loop: clear, draw only visible cells, then draw blinking cursor.
function render() {
var width = canvas.clientWidth;
var height = canvas.clientHeight;
var minX = Math.floor(offsetX / cellWidth) - 1;
var maxX = Math.floor((offsetX + width) / cellWidth) + 1;
var minY = Math.floor(offsetY / cellHeight) - 1;
var maxY = Math.floor((offsetY + height) / cellHeight) + 1;
var now = Date.now();

ctx.clearRect(0, 0, width, height);
ctx.fillStyle = "#000000";

cellMap.forEach(function (value, key) {
var comma = key.indexOf(",");
var worldX;
var worldY;
var drawX;
var drawY;

if (comma <= 0) {
return;
}

worldX = parseInt(key.slice(0, comma), 10);
worldY = parseInt(key.slice(comma + 1), 10);

if (!isFinite(worldX) || !isFinite(worldY)) {
return;
}

if (worldX < minX || worldX > maxX || worldY < minY || worldY > maxY) {
return;
}

drawX = worldX * cellWidth - offsetX;
drawY = worldY * cellHeight - offsetY;
ctx.fillText(value, drawX, drawY + baseline);
});

if (cursorActive && Math.floor(now / 450) % 2 === 0) {
var cursorDrawX = cursorX * cellWidth - offsetX;
var cursorDrawY = cursorY * cellHeight - offsetY;

if (
cursorDrawX >= -cellWidth &&
cursorDrawX <= width + cellWidth &&
cursorDrawY >= -cellHeight &&
cursorDrawY <= height + cellHeight
) {
ctx.fillRect(cursorDrawX, cursorDrawY + 2, 2, cellHeight - 4);
}
}
